### ឧបករណ៍លាក់ព័ត៌មានសម្ងាត់សម្រាប់ការផ្ទៀងផ្ទាត់ (`credentialMasker.ts`)

## កិច្ចសន្យាមូលដ្ឋាន (`base.ts`)

```typescript
class BaseGuardrail {
  enabled: boolean;
  name: string;
  priority: number;

  constructor(name: string, options?: { enabled?: boolean; priority?: number });

  async preCall(payload: unknown, context: GuardrailContext): Promise<GuardrailResult | void>;

  async postCall(response: unknown, context: GuardrailContext): Promise<GuardrailResult | void>;
}

interface GuardrailResult<TValue = unknown> {
  block?: boolean; // true បញ្ឈប់ការដំណើរការខ្សែសង្វាក់ភ្លាមៗ
  message?: string; // បង្ហាញនៅពេលរារាំង
  meta?: Record<string, unknown> | null;
  modifiedPayload?: TValue; // ត្រឡប់ដោយ preCall ដើម្បីសរសេរសំណើឡើងវិញ
  modifiedResponse?: TValue; // ត្រឡប់ដោយ postCall ដើម្បីសរសេរការឆ្លើយតបឡើងវិញ
}

interface GuardrailContext {
  apiKeyInfo?: Record<string, unknown> | null;
  disabledGuardrails?: string[] | null;
  endpoint?: string | null;
  headers?: Headers | Record<string, unknown> | null;
  log?: GuardrailLog | Console | null;
  method?: string | null;
  model?: string | null;
  provider?: string | null;
  signal?: AbortSignal;
  sourceFormat?: string | null;
  stream?: boolean;
  targetFormat?: string | null;
}
```

យន្តការការពារបញ្ជាក់ថា «គ្មានការផ្លាស់ប្តូរ» ដោយត្រឡប់ `void`, `{}`, ឬ
`{ block: false }`។ ការត្រឡប់ `modifiedPayload`/`modifiedResponse` នឹងជំនួស
តម្លៃដែលហូរឆ្លងកាត់ខ្សែសង្វាក់សម្រាប់យន្តការការពារបន្ទាប់។
`signal?: AbortSignal` បញ្ជូនវដ្តជីវិតរបស់អ្នកហៅចូលទៅក្នុងយន្តការការពារ។ ការបោះបង់សំណើគឺជាករណីលើកលែងដែលអនុញ្ញាតឱ្យបន្តពេលបរាជ័យដោយចេតនា៖ ស្ពានភ្ជាប់មេឌៀបញ្ឈប់ការងារ និងសម្អាតធនធាន ដោយមិនដាក់មេឌៀដើមត្រឡប់ទៅគោលដៅដែលដឹងថាមិនគាំទ្រវានោះទេ។

## បញ្ជីចុះឈ្មោះ (`registry.ts`)

អង្គធាតុដែលមានតែមួយ `guardrailRegistry` ផ្តល់៖

- `register(guardrail)` — បន្ថែមយន្តការការពារ (ឬជំនួសតាមឈ្មោះដែលបានធ្វើឱ្យមានទម្រង់ស្តង់ដារ) ហើយ
  តម្រៀបឡើងវិញតាម `priority` ពីតូចទៅធំ។
- `clear()` / `list()` — មុខងារជំនួយសម្រាប់ការគ្រប់គ្រង។
- `runPreCallHooks(payload, context)` — ដំណើរការយន្តការការពារដែលសកម្មម្តងមួយៗ បញ្ជូន
  ទិន្នន័យសំណើឆ្លងកាត់ `modifiedPayload` ហើយឈប់នៅពេលជួប `block: true` ដំបូង។
- `runPostCallHooks(response, context)` — លំហូរដូចគ្នាសម្រាប់ផ្នែកការឆ្លើយតប។
- `resetGuardrailsForTests({ registerDefaults })` — សម្អាតស្ថានភាព ហើយអាច
  ចុះឈ្មោះយន្តការលំនាំដើមឡើងវិញ ដើម្បីឱ្យការធ្វើតេស្តដាច់ដោយឡែកពីគ្នាយ៉ាងស្អាត។

មុខងារដំណើរការទាំងពីរត្រឡប់ `{ blocked, payload|response, results, guardrail?, message? }`
ដែល `results` ជាអារេនៃកំណត់ត្រា `GuardrailExecutionResult` ដែលរួមមាន
វាល `blocked`, `skipped`, `modified`, `error` និង `meta` សម្រាប់យន្តការការពារនីមួយៗ
ដែលមានប្រយោជន៍សម្រាប់ការតាមដានដំណើរការ។

  ការបញ្ចូលបញ្ជាបន្លំក្នុង prompt ដោយហៅ `evaluatePromptInjection()` ដោយខ្លួនឯង ដោយមិន
  បញ្ជូនបញ្ជីបិទណាមួយដែលគ្រប់គ្រងដោយអ្នកហៅឡើយ។ របៀប `block` គឺជា
  ព្រំដែនចាំបាច់ដែលការបិទដោយអ្នកហៅមិនអាចរំលងបាន រួមទាំងនៅលើផ្លូវក្នុងក្រុម chat ផងដែរ
  ([#16413](https://github.com/diegosouzapw/OmniRoute/issues/16413))។ ការបិទ
  `prompt-injection` គ្រាន់តែរំលងយន្តការការពារក្នុងបញ្ជីចុះបញ្ជីប៉ុណ្ណោះ។

### អ្វីដែលទទួលបានការឆ្លើយតបមុនដំណាក់កាលក្រោយការហៅ

នៅលើផ្លូវដែលមិនស្ទ្រីម លំដាប់គឺ៖ ការទាញយកអង្គចងចាំ → ការប្រតិបត្តិឧបករណ៍
ដែលមានស្រាប់ → យន្តការការពារក្រោយការហៅ → ការចាកចេញដោយការទប់ស្កាត់ក្រោយការហៅ → ការរក្សាទុកក្នុងឃ្លាំងសម្ងាត់ផ្អែកលើអត្ថន័យ។

- **ការស្រង់អង្គចងចាំ** និង **ការប្រតិបត្តិឧបករណ៍** ទទួលបានការឆ្លើយតប មុនពេលយន្តការការពារ
  ក្រោយការហៅអាចសរសេរវាឡើងវិញ ឬទប់ស្កាត់វាបាន។ ចាប់តាំងពី
  [#16401](https://github.com/diegosouzapw/OmniRoute/pull/16401) ការស្រង់
  អង្គចងចាំបញ្ជូនអត្ថបទដែលបានស្រង់ចេញឱ្យឆ្លងកាត់ `sanitizePII` នៅពេល
  `PII_RESPONSE_SANITIZATION` ត្រូវបានបើក (ទាំងបែបស្ទ្រីម និងមិនមែនស្ទ្រីម) ប៉ុន្តែ
  ការឆ្លើយតបដែលត្រូវបានទប់ស្កាត់នៅពេលក្រោយដោយយន្តការក្រោយការហៅ នៅតែត្រូវបានបញ្ជូនទៅអង្គចងចាំ និង
  ឧបករណ៍រួចហើយ។
- **ឃ្លាំងសម្ងាត់ផ្អែកលើអត្ថន័យ** រក្សាទុកការឆ្លើយតបបន្ទាប់ពីយន្តការក្រោយការហៅ ដូច្នេះវារក្សាទុក
  ការឆ្លើយតបដែលបានសរសេរឡើងវិញ ហើយការឆ្លើយតបដែលត្រូវបានទប់ស្កាត់មិនត្រូវបានរក្សាទុកក្នុងឃ្លាំងសម្ងាត់ឡើយ។ ធាតុក្នុងឃ្លាំងសម្ងាត់ត្រូវបាន
  កំណត់សោដាច់ដោយឡែកតាមសោ API នីមួយៗ។

## យន្តការការពារផ្ទាល់ខ្លួន

```typescript
import { BaseGuardrail, guardrailRegistry } from "@/lib/guardrails";

class BudgetGuardrail extends BaseGuardrail {
  constructor() {
    super("budget", { priority: 50 });
  }

  async preCall(payload, ctx) {
    if (ctx.apiKeyInfo?.budgetExceeded) {
      return { block: true, message: "Daily budget exceeded" };
    }
    return { block: false };
  }
}

guardrailRegistry.register(new BudgetGuardrail());
```

ជំហាន៖

1. បង្កើត `src/lib/guardrails/myGuardrail.ts` ដែលពង្រីកពី `BaseGuardrail`។
2. អនុវត្ត `preCall` និង/ឬ `postCall`។
3. ចុះឈ្មោះនៅពេលនាំចូល (បន្ថែមពី `registerDefaultGuardrails`) ឬ
   ហៅ `guardrailRegistry.register(...)` នៅពេលដំណើរការ — បញ្ជីចុះឈ្មោះនឹងជំនួស
   យន្តការការពារមុនណាមួយដែលមានឈ្មោះដូចគ្នាក្រោយពេលធ្វើឱ្យមានទម្រង់ស្តង់ដារ។
4. បន្ថែមការធ្វើតេស្តនៅក្រោម `tests/unit/` (ឧទាហរណ៍ដែលមានស្រាប់៖
   `tests/unit/guardrails-registry.test.ts`,
   `tests/unit/prompt-injection-guard.test.ts`,
   `tests/unit/guardrails/visionBridge.test.ts`)។

## វិសាលភាពគ្របដណ្តប់ផ្លូវរបស់យន្តការការពារការចាក់បញ្ចូល និងការធ្វើតេស្តបែបក្រុមវាយប្រហារ (ដំណាក់កាលទី 8 · ប្លុក D)

យន្តការការពារការចាក់បញ្ចូល (`createInjectionGuard` / `withInjectionGuard`) គ្របដណ្តប់គ្រប់ផ្លូវ
ដែលទទួលយកសារបញ្ជារបស់អ្នកប្រើ។ វាគោរពតាម `INJECTION_GUARD_MODE` (លំនាំដើម `warn` = កត់ត្រាតែប៉ុណ្ណោះ;
`block` = ត្រឡប់ HTTP 400 `SECURITY_001`)។

| ប្រភេទ             | ផ្លូវ                                                                                                                                                | របៀបលំនាំដើម |
| ------------------ | ---------------------------------------------------------------------------------------------------------------------------------------------------- | ------------ |
| អត្ថបទ (មានស្រាប់) | `/v1/chat/completions`, `/v1/completions`, `/v1/relay/chat/completions`                                                                              | warn         |
| បង្កើតមាតិកា       | `/v1/messages`, `/v1/responses`, `/v1/images/generations`, `/v1/images/edits`, `/v1/videos/generations`, `/v1/music/generations`, `/v1/audio/speech` | warn         |
| ទិន្នន័យ           | `/v1/embeddings`, `/v1/rerank`, `/v1/search`, `/v1/moderations`                                                                                      | warn         |

ការស្រង់អត្ថបទ (`extractMessageContents`) គ្របដណ្តប់ `messages`/`input`/`prompt`/`query`+`documents`/`instructions`/`system`។

**ការធ្វើតេស្តបែបក្រុមវាយប្រហារ (រៀងរាល់យប់, `nightly-llm-security.yml`)៖** promptfoo ផ្ទៀងផ្ទាត់ថាផ្លូវនីមួយៗទប់ស្កាត់
សំណុំទិន្នន័យ OWASP-LLM ក្នុង `INJECTION_GUARD_MODE=block`; garak ដំណើរការការសាកល្បងវាយប្រហារ (រំលងបើគ្មានព័ត៌មានសម្ងាត់)។
`moderations` ត្រូវបានរួមបញ្ចូលដើម្បីរក្សាភាពស៊ីសង្វាក់គ្នា។ របៀបទប់ស្កាត់គឺជាព្រំដែនសុវត្ថិភាពដែលត្រូវតែអនុវត្ត៖ `evaluatePromptInjection()`
មិនអើពើនឹងបញ្ជីបិទដំណើរការសម្រាប់សំណើនីមួយៗដែល middleware បញ្ជូនឱ្យវា ដូច្នេះ `disabledGuardrails`
**មិន** លើកលែងផ្លូវណាមួយពីការទប់ស្កាត់នេះទេ ([#16413](https://github.com/diegosouzapw/OmniRoute/issues/16413))។

លំហូរការងាររៀងរាល់យប់ (`.github/workflows/nightly-llm-security.yml`, cron + ការចាប់ផ្តើម
ដោយដៃ) មានការងារពីរ៖

- **`promptfoo-guard` (ទប់ស្កាត់)** — ដំណើរការ `promptfoo eval -c promptfooconfig.yaml`
  ជាមួយ `INJECTION_GUARD_MODE=block`។ ករណីវាយប្រហារនីមួយៗ (ឧ. "មិនអើពើនឹង
  សេចក្តីណែនាំពីមុនទាំងអស់…", ការបំបែកការការពារបែប DAN) ផ្ទៀងផ្ទាត់ថាការឆ្លើយតបមាន
  `error.code === "SECURITY_001"` ពោលគឺយន្តការការពារពិតជាបានបដិសេធសំណើ។
- **`garak` (ផ្តល់ព័ត៌មានណែនាំ)** — ដំណើរការ garak `--probes promptinject,dan,leakreplay`
  ទល់នឹងអ៊ីនស្តង់ OmniRoute ក្នុងម៉ាស៊ីនមូលដ្ឋាន (`http://localhost:20128/v1`)។ ដំណើរការតែក្នុងករណីមាន
  ព័ត៌មានសម្ងាត់របស់អ្នកផ្តល់សេវា (`PROMPTFOO_PROVIDER_KEY`); បើគ្មាន វារំលងដោយរលូន ហើយត្រូវបានបន្ថែម
  `|| true` នៅខាងចុង ដូច្នេះវារាយការណ៍ដោយមិនធ្វើឱ្យ CI បរាជ័យ។
